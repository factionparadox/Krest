# Everything Pustota installs on top of Void's base-system, grouped by job.
# Each name is a Void package (checked against void-packages, 31 Oct 2026), or one of
# Pustota's own built from srcpkgs/ (marked below) into Pustota's signed repository.
# They become the dependencies of the pustota-desktop package, so on an installed system
#   xbps-query -x pustota-desktop
# lists them, and removing one is an ordinary xbps-remove.

# kernel and firmware: Void's newest series (7.x) for the Panther Lake Framework 13 Pro
linux-mainline linux-firmware-intel linux-firmware-network sof-firmware intel-ucode
# intel-ucode lives in Void's nonfree repository: this keeps it enabled for updates
void-repo-nonfree
# boot, filesystems, logs, time, memory
grub-x86_64-efi efibootmgr dracut xfsprogs dosfstools exfatprogs ntfs-3g
socklog-void chrony zramen earlyoom xtools
# sessions, permissions, D-Bus
dbus elogind polkit sudo
# network and Bluetooth
NetworkManager networkmanager-dmenu iw bluez libspa-bluetooth bluetuith
# sound (PipeWire, which also serves PulseAudio and ALSA programs)
pipewire wireplumber alsa-pipewire pamixer pulsemixer
# battery and heat
tlp thermald powertop brightnessctl
# the Framework's own tool: battery charge limit, fans, expansion cards (framework_tool)
framework-system
# X11 and Intel graphics
xorg-minimal xinit xrdb setxkbmap xset xsetroot xprop xinput xdpyinfo
xf86-input-libinput mesa-dri mesa-vulkan-intel vulkan-loader intel-media-driver libva-utils
# the desktop: ratpoison, polybar, dmenu, notifications, lock, terminal
ratpoison polybar dmenu dunst libnotify slock xss-lock xbanish xclip maim xterm feh
# X-терминал (Pustota's own, srcpkgs/x-terminal: st with a Plan 9 look) and its font
x-terminal font-go-ttf
# clipboard history (Super+v), night light, firewall, fingerprint reader
clipmenu redshift ufw fprintd
# Japanese text (the IPA fonts: reading, not typing)
ipafont-fonts-otf
# LibreOffice (Ctrl+Super+o): documents, spreadsheets, slides, drawing, maths; British English and
# Russian menus, spelling, hyphenation and thesaurus
libreoffice-writer libreoffice-calc libreoffice-impress libreoffice-draw libreoffice-math
libreoffice-gnome libreoffice-i18n-en-GB libreoffice-i18n-ru
hunspell-en_GB-all hunspell-ru_RU hyphen-en hyphen-ru mythes-en mythes-ru
# weekly TRIM for the SSD (snooze-weekly runs /etc/cron.weekly/fstrim)
snooze run-parts
# files, writing and Plan 9
mc plan9port neovim
# fonts with full Cyrillic
dejavu-fonts-ttf noto-fonts-ttf noto-fonts-emoji terminus-font font-ibm-plex-ttf
# everyday programs
firefox mpv yt-dlp zathura zathura-pdf-mupdf udiskie htop curl wget git rsync unzip p7zip
openssh fwupd fastfetch
# anime in the terminal (Pustota's own package: srcpkgs/ani-cli)
ani-cli
# reading the system: Void's handbook and manuals, offline
void-docs void-docs-browse man-pages mdocml
# Russian kit runtime (the material itself downloads after installing)
python3 kiwix-tools espeak-ng zstd
